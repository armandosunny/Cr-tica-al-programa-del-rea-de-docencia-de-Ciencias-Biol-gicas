normalizar <- function(x) {
  x %>%
    stringi::stri_trans_general("Latin-ASCII") %>%
    stringr::str_to_lower() %>%
    stringr::str_squish()
}

clasificar <- function(nombres_norm, reglas) {
  out <- rep(NA_character_, length(nombres_norm))
  for (i in seq_len(nrow(reglas))) {
    hit <- is.na(out) & stringr::str_detect(nombres_norm, stringr::regex(reglas$patron[i]))
    out[hit] <- reglas$categoria[i]
  }
  out[is.na(out)] <- "Sin clasificar"
  out
}

# ---------------------------------------------------------------------
# 4. COMPOSICIÓN DEL NÚCLEO DISCIPLINAR POR PLAN
#    Obligatorias + actividades (O y AA); sin idiomas ni formación general;
#    sin diplomas técnicos (college)
# ---------------------------------------------------------------------
excluir_nucleo <- c("Idiomas y comunicación", "Formación general y humanidades")

composicion <- datos %>%
  filter(caracter %in% c("O", "AA"),
         !categoria %in% excluir_nucleo,
         nivel != "Diploma técnico (college)") %>%
  count(id_plan, universidad, grupo, categoria, name = "n") %>%
  group_by(id_plan) %>%
  mutate(prop = n / sum(n)) %>%
  ungroup() %>%
  tidyr::complete(tidyr::nesting(id_plan, universidad, grupo), categoria,
                  fill = list(n = 0, prop = 0))

# ---------------------------------------------------------------------
# 5. UAEMex FRENTE A LOS PROMEDIOS NACIONAL E INTERNACIONAL
# ---------------------------------------------------------------------
uaemex_vs <- composicion %>%
  mutate(referencia = case_when(grupo == "UAEMex" ~ "UAEMex",
                                grupo == "México (otras IES)" ~ "Promedio México",
                                TRUE ~ "Promedio internacional")) %>%
  group_by(categoria, referencia) %>%
  summarise(pct = round(100 * mean(prop), 1), .groups = "drop") %>%
  tidyr::pivot_wider(names_from = referencia, values_from = pct) %>%
  mutate(dif_vs_internacional = UAEMex - `Promedio internacional`,
         dif_vs_mexico = UAEMex - `Promedio México`) %>%
  arrange(dif_vs_internacional)

uaemex_vs %>% print(n = 30)

# ---------------------------------------------------------------------
# 6. TEMAS CLAVE EN ASIGNATURAS OBLIGATORIAS
# ---------------------------------------------------------------------
presencia <- datos %>%
  filter(nivel != "Diploma técnico (college)") %>%
  select(id_plan, grupo, caracter, all_of(temas_clave$tema)) %>%
  tidyr::pivot_longer(all_of(temas_clave$tema), names_to = "tema", values_to = "presente") %>%
  group_by(id_plan, grupo, tema) %>%
  summarise(obligatorio   = any(presente & caracter %in% c("O", "AA")),
            solo_optativo = !obligatorio & any(presente & caracter == "Op"),
            .groups = "drop")

temas_resumen <- presencia %>%
  filter(grupo != "UAEMex") %>%
  mutate(ref = if_else(grupo == "México (otras IES)", "pct_Mexico", "pct_Internacional")) %>%
  group_by(tema, ref) %>%
  summarise(pct = round(100 * mean(obligatorio)), .groups = "drop") %>%
  tidyr::pivot_wider(names_from = ref, values_from = pct) %>%
  left_join(presencia %>%
              filter(grupo == "UAEMex") %>%
              transmute(tema, UAEMex = case_when(obligatorio   ~ "Obligatorio",
                                                 solo_optativo ~ "Solo optativo",
                                                 TRUE          ~ "Ausente")),
            by = "tema") %>%
  arrange(desc(pct_Mexico))

temas_resumen %>% print(n = 30)

pares <- tibble::tribble(
  ~antes,               ~despues,
  "Bioquímica",         "Microbiología",
  "Genética",           "Biología molecular",
  "Biología celular",   "Microbiología",
  "Bioquímica",         "Biología molecular",
  "Biología molecular", "Bioinformática y ómicas",
  "Genética",           "Bioinformática y ómicas"
)

secuencias <- purrr::pmap_dfr(pares, function(antes, despues) {
  d <- primer_periodo %>% select(id_plan, a = all_of(antes), b = all_of(despues)) %>%
    filter(!is.na(a), !is.na(b))
  ref <- d %>% filter(id_plan != "UAEMEX")
  uae <- d %>% filter(id_plan == "UAEMEX")
  tibble(secuencia = paste(antes, "antes o a la par de", despues),
         n_planes = nrow(ref),
         pct_planes = round(100 * mean(ref$a <= ref$b)),
         uaemex = if (nrow(uae) == 1) ifelse(uae$a <= uae$b, "Sí", "No") else "Sin dato")
})

# ---------------------------------------------------------------------
# 3. ¿MÉXICO VA ATRASADO O ADELANTADO EN ESTOS TEMAS?
#    Prueba exacta de Fisher por tema (México vs. internacional),
#    con corrección de Benjamini-Hochberg por comparaciones múltiples.
# ---------------------------------------------------------------------
fisher_temas <- estado_temas %>%
  filter(referencia != "UAEMex") %>%
  group_by(tema) %>%
  summarise(p_valor = fisher.test(table(factor(referencia, levels = c("México", "Internacional")),
                                        factor(cualquiera, levels = c(FALSE, TRUE))))$p.value,
            pct_MX = round(100 * mean(cualquiera[referencia == "México"])),
            pct_INT = round(100 * mean(cualquiera[referencia == "Internacional"])),
            .groups = "drop") %>%
  mutate(p_ajustado = round(p.adjust(p_valor, method = "BH"), 3),
         p_valor = round(p_valor, 3),
         conclusion = case_when(p_ajustado < 0.05 & pct_MX > pct_INT ~ "Más frecuente en México",
                                p_ajustado < 0.05 & pct_MX < pct_INT ~ "Más frecuente fuera de México",
                                TRUE ~ "Sin diferencia significativa")) %>%
  arrange(p_valor)

print(fisher_temas, n = 25)

# Simulación (distribución triangular) para expresar la incertidumbre
rtri <- function(n, a, m, b) {
  if (a == b) return(rep(a, n))
  u <- runif(n); f <- (m - a) / (b - a)
  ifelse(u < f, a + sqrt(u * (b - a) * (m - a)), b - sqrt((1 - u) * (b - a) * (b - m)))
}

# 5. ROBUSTEZ DE LA PRIORIDAD
#    5000 combinaciones aleatorias de pesos (Dirichlet) para impacto,
#    facilidad y rapidez. ¿Qué tan seguido queda cada propuesta entre
#    las 3 primeras?
# ---------------------------------------------------------------------
rdirichlet <- function(n, k) { x <- matrix(rgamma(n * k, 1), n, k); x / rowSums(x) }
w <- rdirichlet(5000, 3)
rapidez <- with(indices, 1 - (t_mediana - min(t_mediana)) / diff(range(t_mediana)))

rangos <- purrr::map_dfr(seq_len(nrow(w)), function(i) {
  puntaje <- w[i, 1] * indices$impacto + w[i, 2] * (1 - indices$dificultad) + w[i, 3] * rapidez
  tibble(id = indices$id, rango = rank(-puntaje, ties.method = "min"))
})

robustez <- rangos %>% group_by(id) %>%
  summarise(prob_top3 = mean(rango <= 3), rango_mediano = median(rango), .groups = "drop") %>%
  left_join(propuestas %>% select(id, propuesta), by = "id") %>%
  arrange(desc(prob_top3))

aplicar <- function(cambios) {
  m <- mapa_actual
  if (length(cambios) > 0) m$periodo[match(names(cambios), m$unidad_aprendizaje)] <- cambios
  m
}

evaluar_mapa <- function(cambios, nombre) {
  m <- aplicar(cambios)
  per_todo <- bind_rows(m %>% select(unidad_aprendizaje, periodo),
                        periodos_plan %>% filter(!unidad_aprendizaje %in% m$unidad_aprendizaje))
  conflictos <- m %>%
    filter(seriacion_antecedente != "Ninguna") %>%
    left_join(per_todo %>% rename(seriacion_antecedente = unidad_aprendizaje, periodo_ant = periodo),
              by = "seriacion_antecedente") %>%
    filter(periodo_ant >= periodo) %>%
    transmute(conflicto = paste0(unidad_aprendizaje, " (P", periodo, ") requiere ",
                                 seriacion_antecedente, " (P", periodo_ant, ")"))
  per <- tibble::deframe(m %>% select(unidad_aprendizaje, periodo))
  cumple <- pares %>% mutate(ok = per[antes] <= per[despues])
  carga <- m %>% group_by(periodo) %>% summarise(h = sum(horas))
  tibble(escenario = nombre,
         conflictos_seriacion = nrow(conflictos),
         detalle_conflictos = if (nrow(conflictos) == 0) "Ninguno" else paste(conflictos$conflicto, collapse = "; "),
         secuencias_cumplidas = paste0(sum(cumple$ok), " de ", nrow(cumple)),
         indice_secuencia = round(sum(cumple$peso * cumple$ok) / sum(cumple$peso), 2),
         horas_CB_max_periodo = max(carga$h))
}

# Simulación (distribución triangular) para expresar la incertidumbre
rtri <- function(n, a, m, b) {
  if (a == b) return(rep(a, n))
  u <- runif(n); f <- (m - a) / (b - a)
  ifelse(u < f, a + sqrt(u * (b - a) * (m - a)), b - sqrt((1 - u) * (b - a) * (b - m)))
}

# 5. ROBUSTEZ DE LA PRIORIDAD
#    5000 combinaciones aleatorias de pesos (Dirichlet) para impacto,
#    facilidad y rapidez. ¿Qué tan seguido queda cada propuesta entre
#    las 3 primeras?
# ---------------------------------------------------------------------
rdirichlet <- function(n, k) { x <- matrix(rgamma(n * k, 1), n, k); x / rowSums(x) }
w <- rdirichlet(5000, 3)
rapidez <- with(indices, 1 - (t_mediana - min(t_mediana)) / diff(range(t_mediana)))

rangos <- purrr::map_dfr(seq_len(nrow(w)), function(i) {
  puntaje <- w[i, 1] * indices$impacto + w[i, 2] * (1 - indices$dificultad) + w[i, 3] * rapidez
  tibble(id = indices$id, rango = rank(-puntaje, ties.method = "min"))
})

robustez <- rangos %>% group_by(id) %>%
  summarise(prob_top3 = mean(rango <= 3), rango_mediano = median(rango), .groups = "drop") %>%
  left_join(propuestas %>% select(id, propuesta), by = "id") %>%
  arrange(desc(prob_top3))

print(robustez, width = Inf)

aplicar <- function(cambios) {
  m <- mapa_actual
  if (length(cambios) > 0) m$periodo[match(names(cambios), m$unidad_aprendizaje)] <- cambios
  m
}

evaluar_mapa <- function(cambios, nombre) {
  m <- aplicar(cambios)
  per_todo <- bind_rows(m %>% select(unidad_aprendizaje, periodo),
                        periodos_plan %>% filter(!unidad_aprendizaje %in% m$unidad_aprendizaje))
  conflictos <- m %>%
    filter(seriacion_antecedente != "Ninguna") %>%
    left_join(per_todo %>% rename(seriacion_antecedente = unidad_aprendizaje, periodo_ant = periodo),
              by = "seriacion_antecedente") %>%
    filter(periodo_ant >= periodo) %>%
    transmute(conflicto = paste0(unidad_aprendizaje, " (P", periodo, ") requiere ",
                                 seriacion_antecedente, " (P", periodo_ant, ")"))
  per <- tibble::deframe(m %>% select(unidad_aprendizaje, periodo))
  cumple <- pares %>% mutate(ok = per[antes] <= per[despues])
  carga <- m %>% group_by(periodo) %>% summarise(h = sum(horas))
  tibble(escenario = nombre,
         conflictos_seriacion = nrow(conflictos),
         detalle_conflictos = if (nrow(conflictos) == 0) "Ninguno" else paste(conflictos$conflicto, collapse = "; "),
         secuencias_cumplidas = paste0(sum(cumple$ok), " de ", nrow(cumple)),
         indice_secuencia = round(sum(cumple$peso * cumple$ok) / sum(cumple$peso), 2),
         horas_CB_max_periodo = max(carga$h))
}

# Simulación (distribución triangular) para expresar la incertidumbre
rtri <- function(n, a, m, b) {
  if (a == b) return(rep(a, n))
  u <- runif(n); f <- (m - a) / (b - a)
  ifelse(u < f, a + sqrt(u * (b - a) * (m - a)), b - sqrt((1 - u) * (b - a) * (b - m)))
}

  select(id, sim) %>% tidyr::unnest(sim) %>% rename(semestres = sim)

# ---------------------------------------------------------------------
# 5. ROBUSTEZ DE LA PRIORIDAD
#    5000 combinaciones aleatorias de pesos (Dirichlet) para impacto,
#    facilidad y rapidez. ¿Qué tan seguido queda cada propuesta entre
#    las 3 primeras?
# ---------------------------------------------------------------------
rdirichlet <- function(n, k) { x <- matrix(rgamma(n * k, 1), n, k); x / rowSums(x) }
w <- rdirichlet(5000, 3)
rapidez <- with(indices, 1 - (t_mediana - min(t_mediana)) / diff(range(t_mediana)))

rangos <- purrr::map_dfr(seq_len(nrow(w)), function(i) {
  puntaje <- w[i, 1] * indices$impacto + w[i, 2] * (1 - indices$dificultad) + w[i, 3] * rapidez
  tibble(id = indices$id, rango = rank(-puntaje, ties.method = "min"))
})

robustez <- rangos %>% group_by(id) %>%
  summarise(prob_top3 = mean(rango <= 3), rango_mediano = median(rango), .groups = "drop") %>%
  left_join(propuestas %>% select(id, propuesta), by = "id") %>%
  arrange(desc(prob_top3))

print(robustez, width = Inf)
readr::write_csv(robustez, file.path(dir_res, "tabla_robustez_prioridad.csv"))

figM <- ggplot(robustez, aes(x = 100 * prob_top3,
                             y = forcats::fct_reorder(paste(id, propuesta), prob_top3))) +
  geom_col(fill = "#2166AC", width = 0.65) +
  geom_text(aes(label = paste0(round(100 * prob_top3), "%  (rango mediano ", rango_mediano, ")")),
            hjust = -0.05, size = 3.2) +
  scale_x_continuous(limits = c(0, 130), breaks = seq(0, 100, 25)) +
  labs(x = "% de combinaciones de pesos en que la propuesta queda entre las 3 prioritarias", y = NULL,
       title = "Robustez de la prioridad de cada propuesta",
       subtitle = "5,000 combinaciones aleatorias de pesos para impacto, facilidad y rapidez") +
  theme_minimal(base_size = 11) + theme(plot.title.position = "plot")

guardar(figM, "figM_robustez_prioridad.png", 10, 6)

aplicar <- function(cambios) {
  m <- mapa_actual
  if (length(cambios) > 0) m$periodo[match(names(cambios), m$unidad_aprendizaje)] <- cambios
  m
}

evaluar_mapa <- function(cambios, nombre) {
  m <- aplicar(cambios)
  per_todo <- bind_rows(m %>% select(unidad_aprendizaje, periodo),
                        periodos_plan %>% filter(!unidad_aprendizaje %in% m$unidad_aprendizaje))
  conflictos <- m %>%
    filter(seriacion_antecedente != "Ninguna") %>%
    left_join(per_todo %>% rename(seriacion_antecedente = unidad_aprendizaje, periodo_ant = periodo),
              by = "seriacion_antecedente") %>%
    filter(periodo_ant >= periodo) %>%
    transmute(conflicto = paste0(unidad_aprendizaje, " (P", periodo, ") requiere ",
                                 seriacion_antecedente, " (P", periodo_ant, ")"))
  per <- tibble::deframe(m %>% select(unidad_aprendizaje, periodo))
  cumple <- pares %>% mutate(ok = per[antes] <= per[despues])
  carga <- m %>% group_by(periodo) %>% summarise(h = sum(horas))
  tibble(escenario = nombre,
         conflictos_seriacion = nrow(conflictos),
         detalle_conflictos = if (nrow(conflictos) == 0) "Ninguno" else paste(conflictos$conflicto, collapse = "; "),
         secuencias_cumplidas = paste0(sum(cumple$ok), " de ", nrow(cumple)),
         indice_secuencia = round(sum(cumple$peso * cumple$ok) / sum(cumple$peso), 2),
         horas_CB_max_periodo = max(carga$h))
}

boot_prev <- function(x, B = 2000) {
  bs <- replicate(B, mean(sample(x, replace = TRUE)))
  tibble(prev = mean(x), li = quantile(bs, 0.025), ls = quantile(bs, 0.975))
}

  group_modify(~ boot_prev(.x$obligatoria)) %>%
  ungroup() %>%
  left_join(presencia %>% filter(grupo == "UAEMex") %>% select(subarea, UAEMex = estado), by = "subarea")

evaluar_mapa <- function(cambios, nombre) {
  m <- aplicar(cambios)
  per_todo <- bind_rows(m %>% select(unidad_aprendizaje, periodo),
                        periodos_plan %>% filter(!unidad_aprendizaje %in% m$unidad_aprendizaje))
  conflictos <- m %>%
    filter(seriacion_antecedente != "Ninguna") %>%
    left_join(per_todo %>% rename(seriacion_antecedente = unidad_aprendizaje, periodo_ant = periodo),
              by = "seriacion_antecedente") %>%
    filter(periodo_ant >= periodo)
  per <- tibble::deframe(m %>% select(unidad_aprendizaje, periodo))
  cumple <- pares %>% mutate(ok = per[antes] <= per[despues])
  tibble(escenario = nombre, conflictos_seriacion = nrow(conflictos),
         indice_secuencia = round(sum(cumple$peso * cumple$ok) / sum(cumple$peso), 2))
}
